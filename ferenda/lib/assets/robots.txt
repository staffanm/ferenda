# lagen.nu's crawl policy. Every page a reader can open is crawlable. What is
# kept out is machinery, not content: the JSON API and its docs, the MCP
# endpoint, the search shell (an unbounded query space), the editor login and
# the bulk dumps. Nothing here says noindex: on the preview host
# (ferenda.lagen.nu) that is nginx's X-Robots-Tag header, which a crawler only
# sees by fetching the page -- so this file must never Disallow the pages.

Disallow: /mcp
Disallow: /docs
Disallow: /sok
Disallow: /admin/
